\section*{Problem 2}
Consider the Least Absolute Deviation (LAD) problem:
\begin{equation}\label{eq:lad}
\Min_{\vx\in \RR^n} \|\vA\vx - \vb\|_1
\end{equation}
where $\vA\in\RR^{m\times n}$ and $\vb\in\RR^m$ are given. The LAD problem is equivalent to
\begin{equation}\label{eq:lad2}
\Min_{\vx\in \RR^n, \vz\in\RR^m} \|\vz\|_1, \st\ \vA\vx -\vz-\vb=\vzero.
\end{equation}
The augmented Lagrangian function of \eqref{eq:lad2} is
$$\cL_\rho(\vx,\vz,\vy)=\|\vz\|_1+\vy^\top(\vA\vx -\vz-\vb)+\frac{\rho}{2}\|\vA\vx -\vz-\vb\|^2,$$
where $\rho>0$ is the penalty parameter, and $\vy\in\RR^m$ is the multiplier vector. The iterative update schemes of ADMM on solving \eqref{eq:lad2} is
\begin{subequations}
\begin{align}
\vx^{k+1}=&\argmin_{\vx\in\RR^n} \cL_\rho(\vx,\vz^{k},\vy^{k}),\label{eq:lad2-x}\\
\vz^{k+1}=&\argmin_{\vz\in\RR^m} \cL_\rho(\vx^{k+1},\vz,\vy^{k}),\label{eq:lad2-z}\\
\vy^{k+1}=&\vy^k+\rho(\vA\vx^{k+1}-\vz^{k+1}-\vb)
\end{align}
\end{subequations}

\begin{enumerate}
\item Assume $\vA$ has column full rank, i.e., $\vA^\top \vA$ is nonsingular. Give the closed-form solution of $\vx^{k+1}$ in the update \eqref{eq:lad2-x}. The solution should involve the inverse of $\vA^\top \vA$.

\item Recall that we have derived the proximal mapping of $\lambda\|\cdot\|_1$ for any $\lambda>0$ in class. Let's denote it as $\prox_{\lambda\|\cdot\|_1}$. Use the proximal mapping to derive a closed form solution of $\vz^{k+1}$ in the update \eqref{eq:lad2-z}. In the solution, $\lambda$ should be $1/\rho$.

\item Suppose $(\vx^{k+1},\vz^{k+1},\vy^{k+1})$ is the output. At this point, give the violation of primal and dual feasibility.

\item Use the instructor's provided file \verb|ADMM_LAD.m| to write a Matlab function \verb|ADMM_LAD| with input $\vA$, $\vb$, initial vector $\vx0$, stopping tolerance \verb|tol|, maximum number of iterations \verb|maxit|, and penalty parameter $\rho>0$. Also test your function by running the provided test file \verb|test_ADMM_LAD.m| and compare to the instructor's function. Print your code and the results you get.
\end{enumerate}
